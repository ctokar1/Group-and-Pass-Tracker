        // --- DUAL STORAGE ENGINE (localStorage + IndexedDB) ---
        const DB_NAME = 'TigerGroupCraftDB';
        const DB_VERSION = 1;
        const STORE_NAME = 'app_state_store';
        const STORAGE_KEY = 'tiger_groupcraft_state_v2';

        function initIndexedDB() {
            return new Promise((resolve) => {
                if (!window.indexedDB) return resolve(null);
                try {
                    const request = indexedDB.open(DB_NAME, DB_VERSION);
                    request.onupgradeneeded = function(e) {
                        const db = e.target.result;
                        if (!db.objectStoreNames.contains(STORE_NAME)) {
                            db.createObjectStore(STORE_NAME);
                        }
                    };
                    request.onsuccess = function(e) { resolve(e.target.result); };
                    request.onerror = function(e) { resolve(null); };
                } catch(e) {
                    resolve(null);
                }
            });
        }

        async function saveState() {
            // 1. Save to localStorage
            try {
                localStorage.setItem(STORAGE_KEY, JSON.stringify(appState));
            } catch (e) {
                console.warn("localStorage save error", e);
            }

            // 2. Save to IndexedDB (persists across iframe origin changes on Google Sites)
            try {
                const db = await initIndexedDB();
                if (db) {
                    const tx = db.transaction(STORE_NAME, 'readwrite');
                    const store = tx.objectStore(STORE_NAME);
                    store.put(JSON.parse(JSON.stringify(appState)), 'current_state');
                }
            } catch (e) {
                console.warn("IndexedDB save error", e);
            }

            updateStorageStatusBadge(true);
        }

        async function loadState() {
            let loaded = false;

            // 1. Primary: Try localStorage
            try {
                const saved = localStorage.getItem(STORAGE_KEY);
                if (saved) {
                    const parsed = JSON.parse(saved);
                    if (parsed && parsed.sections && parsed.sections.length > 0) {
                        appState = parsed;
                        loaded = true;
                    }
                }
            } catch (e) {
                console.warn("localStorage load error", e);
            }

            // 2. Fallback: Try IndexedDB if localStorage was cleared/sandboxed
            if (!loaded) {
                try {
                    const db = await initIndexedDB();
                    if (db) {
                        const tx = db.transaction(STORE_NAME, 'readonly');
                        const store = tx.objectStore(STORE_NAME);
                        const req = store.get('current_state');
                        await new Promise((resolve) => {
                            req.onsuccess = function() {
                                if (req.result && req.result.sections && req.result.sections.length > 0) {
                                    appState = req.result;
                                    loaded = true;
                                }
                                resolve();
                            };
                            req.onerror = function() { resolve(); };
                        });
                    }
                } catch (e) {
                    console.warn("IndexedDB load error", e);
                }
            }

            updateStorageStatusBadge(loaded);
        }

        window.triggerGroupGeneration = function triggerGroupGeneration() {
            const sec = getActiveSection();
            if (!sec) return;

            const presentStudents = sec.roster.filter(s => s.present);
            if (presentStudents.length === 0) {
                showToast("No students marked present in roster!", "error");
                return;
            }

            const shuffled = [...presentStudents].sort(() => Math.random() - 0.5);
            const totalCount = shuffled.length;

            let qSize = sec.settings.quickSize || 0;
            let qCount = sec.settings.quickCount || 0;
            let forceExact = sec.settings.forceExactCount;

            let numGroups = 0;
            let targetCapacity = 0;

            if (qCount > 0 && qSize > 0) {
                // DUAL CONSTRAINT: Exact N groups, filled up to S capacity per group
                numGroups = qCount;
                targetCapacity = qSize;
            } else if (qCount > 0) {
                // EXACT GROUP COUNT ONLY: Evenly divide total kids into N groups
                numGroups = qCount;
                targetCapacity = Math.ceil(totalCount / numGroups);
            } else if (qSize > 0) {
                // GROUP SIZE ONLY: Calculate needed groups for target size
                numGroups = Math.ceil(totalCount / qSize);
                targetCapacity = qSize;
            } else {
                // DEFAULT: Groups of 4
                qSize = 4;
                numGroups = Math.ceil(totalCount / qSize);
                targetCapacity = qSize;
            }

            numGroups = Math.max(1, Math.min(numGroups, 8)); // Cap between 1 and 8 color tables

            // Initialize station group objects
            let groups = [];
            for (let i = 0; i < numGroups; i++) {
                const colorDef = COLOR_DEFINITIONS[i % COLOR_DEFINITIONS.length];
                groups.push({
                    id: `grp_${i + 1}`,
                    colorIndex: i,
                    name: colorDef.name,
                    colorClass: colorDef.color,
                    borderClass: colorDef.border,
                    bgClass: colorDef.bg,
                    textClass: colorDef.text,
                    students: []
                });
            }

            let studentIdx = 0;

            if (qCount > 0 && qSize > 0) {
                // Fill groups 1..N up to targetCapacity (e.g. 5) strictly first
                for (let g = 0; g < groups.length; g++) {
                    while (groups[g].students.length < targetCapacity && studentIdx < totalCount) {
                        groups[g].students.push(shuffled[studentIdx++]);
                    }
                }
                // If there are overflow students beyond targetCapacity * numGroups,
                // randomly distribute overflow students into existing groups without creating new ones
                while (studentIdx < totalCount) {
                    const randomGrp = groups[Math.floor(Math.random() * groups.length)];
                    randomGrp.students.push(shuffled[studentIdx++]);
                }
            } else {
                // EVEN BALANCED DISTRIBUTION across numGroups
                // e.g. 14 students across 4 groups -> 4, 4, 3, 3
                let currentGroup = 0;
                while (studentIdx < totalCount) {
                    groups[currentGroup % numGroups].students.push(shuffled[studentIdx++]);
                    currentGroup++;
                }
            }

            sec.groups = groups;
            saveState();
            renderGroups();
            if (document.getElementById('projectorOverlay')?.classList.contains('hidden') === false) {
                renderProjectorGroups();
            }
            showToast("Shuffled color groups!");
        };

        // --- DRAG AND DROP HANDLERS ---
        window.handleDragStart = function handleDragStart(e, studentId, fromGroupId) {
            e.dataTransfer.setData('text/plain', JSON.stringify({ studentId, fromGroupId }));
        };

        window.handleDragOver = function handleDragOver(e) {
            e.preventDefault();
        };

        window.handleDrop = function handleDrop(e, toGroupId) {
            e.preventDefault();
            const dataStr = e.dataTransfer.getData('text/plain');
            if (!dataStr) return;
            try {
                const { studentId, fromGroupId } = JSON.parse(dataStr);
                moveStudentBetweenGroups(studentId, fromGroupId, toGroupId);
            } catch (err) {
                console.error("Drop error", err);
            }
        };

        function moveStudentBetweenGroups(studentId, fromGroupId, toGroupId) {
            if (fromGroupId === toGroupId) return;
            const sec = getActiveSection();
            if (!sec) return;

            const fromGrp = sec.groups.find(g => g.id === fromGroupId);
            const toGrp = sec.groups.find(g => g.id === toGroupId);
            if (!fromGrp || !toGrp) return;

            const studentIdx = fromGrp.students.findIndex(s => s.id === studentId);
            if (studentIdx === -1) return;

            const [student] = fromGrp.students.splice(studentIdx, 1);
            toGrp.students.push(student);

            saveState();
            renderGroups();
            if (document.getElementById('projectorOverlay')?.classList.contains('hidden') === false) {
                renderProjectorGroups();
            }
        }

        window.removeStudentFromGroup = function removeStudentFromGroup(groupId, studentId) {
            const sec = getActiveSection();
            if (!sec) return;
            const grp = sec.groups.find(g => g.id === groupId);
            if (grp) {
                grp.students = grp.students.filter(s => s.id !== studentId);
                saveState();
                renderGroups();
            }
        };

                subtitle.textContent = text;

        window.toggleTimer = function toggleTimer() {
            if (isTimerRunning) {
                clearInterval(timerInterval);
                isTimerRunning = false;
                document.getElementById('timerPlayBtn').textContent = "▶️";
            } else {
                isTimerRunning = true;
                document.getElementById('timerPlayBtn').textContent = "⏸️";
                timerInterval = setInterval(() => {
                    if (timerSecondsRemaining > 0) {
                        timerSecondsRemaining--;
                        updateTimerDisplay();
                    } else {
                        clearInterval(timerInterval);
                        isTimerRunning = false;
                        document.getElementById('timerPlayBtn').textContent = "▶️";
                        showToast("🔔 Time's Up!");
                    }
                }, 1000);
            }
        };

        window.resetTimer = function resetTimer() {
            clearInterval(timerInterval);
            isTimerRunning = false;
            timerSecondsRemaining = 300;
            document.getElementById('timerPlayBtn').textContent = "▶️";
            updateTimerDisplay();
        };

        function updateTimerDisplay() {
            const m = Math.floor(timerSecondsRemaining / 60);
            const s = timerSecondsRemaining % 60;

            const clock = document.getElementById('projectorTimerClock');
            if (clock) clock.textContent = formatted;

        window.executeSpotlightPick = function executeSpotlightPick() {
            const sec = getActiveSection();
            if (!sec || !sec.groups || sec.groups.length === 0) {
                showToast("Generate groups first before picking a student!", "error");
                return;
            }

            let allAssignedStudents = [];
            sec.groups.forEach(g => {
                g.students.forEach(st => {
                    allAssignedStudents.push({ student: st, group: g });
                });
            });

            if (allAssignedStudents.length === 0) {
                showToast("No active students found in groups!", "error");
                return;
            }

            const nameDisplay = document.getElementById('spotlightNameDisplay');
            const groupDisplay = document.getElementById('spotlightGroupDisplay');

            let counter = 0;
            const maxCycles = 20;
            const interval = setInterval(() => {
                const rand = allAssignedStudents[Math.floor(Math.random() * allAssignedStudents.length)];
                if (nameDisplay) nameDisplay.textContent = rand.student.name;
                if (groupDisplay) groupDisplay.textContent = `Station: ${rand.group.name}`;
                counter++;

                if (counter >= maxCycles) {
                    clearInterval(interval);
                    const winner = allAssignedStudents[Math.floor(Math.random() * allAssignedStudents.length)];
                    if (nameDisplay) nameDisplay.textContent = winner.student.name;
                    if (groupDisplay) groupDisplay.textContent = `🏆 Winner from ${winner.group.name}!`;
                    
                    // Highlight on projector view if active
                    const projBanner = document.getElementById('projectorSpotlightBanner');
                    if (projBanner) {
                        projBanner.textContent = `🏆 Spotlight Winner: ${winner.student.name} (${winner.group.name})`;
                        projBanner.classList.remove('hidden');
                    }

                    showToast(`Spotlight: ${winner.student.name}!`);
                }
            }, 80);
        };


        // --- HALL PASS TRACKER ---
        window.issuePass = function issuePass(stId) {
            const sec = getActiveSection();
            if (!sec) return;
            const st = sec.roster.find(s => s.id === stId);
            if (!st) return;

            if (st.passesUsed >= MAX_QUARTER_PASSES) {
                showToast(`${st.name} has reached the pass limit (${MAX_QUARTER_PASSES}/${MAX_QUARTER_PASSES})!`, "error");
                return;
            }

            st.passesUsed++;
            st.outTimestamp = Date.now();
            saveState();
            renderPasses();
            showToast(`Issued pass to ${st.name} (${st.passesUsed}/${MAX_QUARTER_PASSES})`);
        };

        window.returnStudentFromPass = function returnStudentFromPass(stId) {
            const sec = getActiveSection();
            if (!sec) return;
            const st = sec.roster.find(s => s.id === stId);
            if (!st) return;
            st.outTimestamp = null;
            saveState();
            renderPasses();
            showToast(`${st.name} returned to class`);
        };

        window.adjustPassCount = function adjustPassCount(stId, delta) {
            const sec = getActiveSection();
            if (!sec) return;
            const st = sec.roster.find(s => s.id === stId);
            if (!st) return;
            const newCount = st.passesUsed + delta;
            if (newCount >= 0 && newCount <= MAX_QUARTER_PASSES) {
                st.passesUsed = newCount;
                saveState();
                renderPasses();
            }
        };

        window.renderPasses = function renderPasses() {
            const sec = getActiveSection();
            const grid = document.getElementById('passTrackerGrid');
            const outBanner = document.getElementById('currentlyOutBanner');
            const outList = document.getElementById('currentlyOutList');
            const outBadge = document.getElementById('currentlyOutCountBadge');
            const searchInput = document.getElementById('passSearchInput');
            const query = (searchInput ? searchInput.value : '').toLowerCase().trim();

            if (!sec || !grid) return;

            // Sort Alphabetically A-Z
            const sortedRoster = [...sec.roster].sort((a, b) => a.name.localeCompare(b.name));
            const filteredRoster = sortedRoster.filter(st => st.name.toLowerCase().includes(query));

            // Currently Out Live Banner
            const currentlyOut = sec.roster.filter(st => st.outTimestamp);
            if (currentlyOut.length > 0) {
                if (outBanner) outBanner.classList.remove('hidden');
                if (outBadge) outBadge.textContent = `${currentlyOut.length} Out`;
                if (outList) {
                    outList.innerHTML = currentlyOut.map(st => {
                        const elapsedMins = Math.floor((Date.now() - st.outTimestamp) / 60000);
                        return `
                            <div class="bg-slate-900 border border-amber-400/80 rounded-xl p-3 flex items-center justify-between shadow">
                                <div>
                                    <span class="font-black text-xs text-white block">${escapeHtml(st.name)}</span>
                                    <span class="text-[10px] text-amber-300 font-bold">Out ${elapsedMins} min ago</span>
                                </div>
                                <button onclick="window.returnStudentFromPass('${st.id}')" class="px-2.5 py-1 bg-emerald-500 hover:bg-emerald-400 text-black font-black text-xs rounded-lg shadow">
                                    🟢 Return
                                </button>
                            </div>
                        `;
                    }).join('');
                }
            } else {
                if (outBanner) outBanner.classList.add('hidden');
            }

            // Compact Alphabetical Pass Grid
            if (filteredRoster.length === 0) {
                grid.innerHTML = `<div class="col-span-full py-8 text-center text-slate-500 font-bold text-xs">No matching students found.</div>`;
                return;
            }

            grid.innerHTML = filteredRoster.map(st => {
                const isMax = st.passesUsed >= MAX_QUARTER_PASSES;
                const isOut = !!st.outTimestamp;

                let tokensHtml = '';
                for (let i = 1; i <= MAX_QUARTER_PASSES; i++) {
                    if (i <= st.passesUsed) {
                        tokensHtml += `<span title="Pass Used" class="text-xs">🎟️</span>`;
                    } else {
                        tokensHtml += `<span title="Pass Available" class="text-xs text-slate-700">⚪</span>`;
                    }
                }

                return `
                    <div class="bg-slate-950 border ${isOut ? 'border-amber-400 bg-amber-950/20' : isMax ? 'border-rose-500/60 bg-rose-950/20' : 'border-slate-800'} rounded-xl p-3 flex flex-col justify-between space-y-2 shadow hover:border-slate-700 transition-all">
                        <div class="flex items-center justify-between">
                            <span class="font-extrabold text-xs text-white truncate" title="${escapeHtml(st.name)}">${escapeHtml(st.name)}</span>
                            <div class="flex items-center gap-1">
                                <button onclick="window.adjustPassCount('${st.id}', -1)" ${st.passesUsed <= 0 ? 'disabled' : ''} class="w-4 h-4 bg-slate-800 hover:bg-slate-700 disabled:opacity-30 text-amber-300 text-[10px] font-black rounded flex items-center justify-center">-</button>
                                <button onclick="window.adjustPassCount('${st.id}', 1)" ${st.passesUsed >= MAX_QUARTER_PASSES ? 'disabled' : ''} class="w-4 h-4 bg-slate-800 hover:bg-slate-700 disabled:opacity-30 text-amber-300 text-[10px] font-black rounded flex items-center justify-center">+</button>
                            </div>
                        </div>

                        <div class="flex items-center justify-between border-t border-b border-slate-800/80 py-1 px-0.5">
                            <div class="flex items-center gap-0.5">${tokensHtml}</div>
                            <span class="text-[10px] font-black ${isMax ? 'text-rose-400' : 'text-amber-300'}">
                                ${st.passesUsed}/${MAX_QUARTER_PASSES}
                            </span>
                        </div>

                        ${isOut ? `
                            <button onclick="window.returnStudentFromPass('${st.id}')" class="w-full py-1 bg-emerald-500 hover:bg-emerald-400 text-black font-extrabold text-[11px] rounded-lg shadow">
                                🟢 Return
                            </button>
                        ` : isMax ? `
                            <button disabled class="w-full py-1 bg-rose-950 text-rose-400 border border-rose-500/40 font-extrabold text-[10px] rounded-lg cursor-not-allowed">
                                🚫 Limit Reached
                            </button>
                        ` : `
                            <button onclick="window.issuePass('${st.id}')" class="w-full py-1 bg-slate-800 hover:bg-amber-400 hover:text-black text-amber-300 font-extrabold text-[11px] rounded-lg border border-amber-500/30 transition-all">
                                🎟️ Issue Pass
                            </button>
                        `}
                    </div>
                `;
            }).join('');
        };

        window.exportGroupsCSV = function exportGroupsCSV() {
            const sec = getActiveSection();
            if (!sec || !sec.groups) return;

            let csv = "Group Name,Student Name\n";

            const a = document.createElement('a');

            a.download = `${sec.name.replace(/[^a-z0-9]/gi, '_')}_Groups.csv`;
            a.click();
            URL.revokeObjectURL(url);
            showToast("CSV file exported!");
        };

        function copyTextToClipboard(text, successMsg) {
            const textArea = document.createElement("textarea");
            textArea.value = text;
            textArea.style.position = "fixed";
            textArea.style.opacity = "0";
            document.body.appendChild(textArea);
            textArea.focus();
            textArea.select();
            try {
                document.execCommand('copy');
                showToast(successMsg);
            } catch (err) {
                showToast("Copy failed", "error");
            }
            document.body.removeChild(textArea);
        }

        window.exportDataBackup = function exportDataBackup() {
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(appState, null, 2));
            const downloadAnchor = document.createElement('a');
            downloadAnchor.setAttribute("href", dataStr);
            downloadAnchor.setAttribute("download", `Tiger_GroupCraft_Backup_${new Date().toISOString().slice(0,10)}.json`);
            document.body.appendChild(downloadAnchor);

        window.importDataBackup = function importDataBackup(e) {
            const file = e.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = function(evt) {
                try {
                    const imported = JSON.parse(evt.target.result);
                    if (imported && imported.sections) {
                        appState = imported;
                        saveState();
                        renderSectionDropdown();
                        renderRoster();
                        renderGroups();
                        renderPasses();
                        syncSettingsUI();
                        showToast("Data backup successfully restored!");
                    }
                } catch (err) {
                    showToast("Invalid backup JSON file!", "error");
                }
            };
            reader.readAsText(file);
        };
